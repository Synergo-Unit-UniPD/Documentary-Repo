\section{Casi d'Uso}

\subsection{Attori}
\begin{itemize}
	\item \textbf{Utente}: Attore primario. Rappresenta 
	l'utilizzatore finale che interagisce con 
	l'interfaccia dell'editor.
	\item \textbf{LLM (Large Language Model)}: Attore 
	secondario (sistema esterno). 
	Fornisce i servizi di analisi e generazione testo.
\end{itemize}
