\section{Casi d'Uso}

\subsection{Attori}
\begin{itemize}
	\item \textbf{Utente}: Attore primario. Rappresenta l'utilizzatore finale che interagisce con l'interfaccia dell'applicazione. In base al capitolato e ai verbali esterni, il profilo di riferimento è un utente giovane — tipicamente uno studente — che utilizza l'applicazione per prendere appunti e desidera supporto nell'elaborazione tramite LLM (VE-5.3).
	\item \textbf{LLM (Large Language Model)}: Attore secondario (sistema esterno). Fornisce i servizi di analisi e generazione testo tramite API compatibile con lo standard OpenAI. I modelli di riferimento sono Gemma, ChatGPT-OSS e Qwen, messi a disposizione dall'azienda proponente (VE-1.1).
\end{itemize}

% UC1 - Creazione di una nuova nota vuota
\subsubsection{UC 1: Creazione di una nuova nota vuota}

% use case diagram
\begin{figure}[H]
    \centering
    \includegraphics[width=0.6\textwidth]{creazione_nota.png}
    \caption{UC1 – Creazione di una nuova nota}
\end{figure}

\begin{itemize}
    \item \textbf{Attore Principale}: Utente
    \item \textbf{Precondizioni}: Il sistema è avviato e l'interfaccia è correttamente caricata.
    \item \textbf{Scenario Principale}:
    \begin{enumerate}
        \item L'utente attiva il comando ``Nuova Nota''.
        \item Il sistema richiede l'inserimento di un titolo per la nota.
        \item L'utente inserisce il titolo desiderato e conferma.
        \item Il sistema crea la nota e la apre nell'editor.
    \end{enumerate}
    \item \textbf{Postcondizioni}: Una nuova nota con titolo è aperta e attiva nell'editor.
    \item \textbf{Trigger}: L'utente desidera iniziare una nuova sessione di scrittura.
\end{itemize}

    \item \textbf{Trigger}: L'utente decide di rimuovere una nota non più necessaria.
\end{itemize}
