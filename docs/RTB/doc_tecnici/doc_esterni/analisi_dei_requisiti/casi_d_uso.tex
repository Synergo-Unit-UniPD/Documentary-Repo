\section{Casi d'Uso}

\subsection{Attori}
\begin{itemize}
	\item \textbf{Utente}: Attore primario. Rappresenta l'utilizzatore finale che interagisce con l'interfaccia dell'editor.
	\item \textbf{LLM (Large Language Model)}: Attore secondario (sistema esterno). Fornisce i servizi di analisi e generazione testo.
\end{itemize}

\subsubsection{UC 1: Creazione di una nuova nota vuota}

% use case diagram
\begin{figure}[H]
    \centering
    \includegraphics[width=0.6\textwidth]{creazione_nota.png}
    \caption{UC1 – Creazione di una nuova nota}
\end{figure}

\begin{itemize}
    \item \textbf{Attore Principale}: Utente
    \item \textbf{Scenario Principale}: 
    \begin{enumerate}
        \item L'utente attiva il comando "Nuova Nota";
        \item Il sistema crea una nuova nota vuota.
    \end{enumerate}
    \item \textbf{Precondizioni}: Il sistema è avviato e l'interfaccia è correttamente caricata.
    \item \textbf{Postcondizioni}: Viene creata una nota vuota.
    \item \textbf{Trigger}: L'utente desidera iniziare una nuova sessione di scrittura.
\end{itemize}
