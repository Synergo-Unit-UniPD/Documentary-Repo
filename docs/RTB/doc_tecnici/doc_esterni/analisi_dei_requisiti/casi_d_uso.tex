\section{Casi d'Uso}

\subsection{Attori}
\begin{itemize}
	\item \textbf{Utente}: Attore\textsubscript{G} primario $\rightarrow$ Rappresenta l'utilizzatore finale che interagisce con l'interfaccia dell'applicazione. 
    In base al capitolato e ai verbali esterni, il profilo di riferimento è un utente giovane, tipicamente uno studente, che utilizza l'applicazione per prendere appunti e desidera supporto nell'elaborazione tramite LLM (VE-5.3).
	\item \textbf{LLM (Large Language Model\textsubscript{G})}: Attore secondario (sistema esterno) $\rightarrow$ Fornisce i servizi di analisi e generazione testo tramite API compatibile con lo standard OpenAI. I modelli di riferimento sono Gemma, ChatGPT-OSS e Qwen, messi a disposizione dall'azienda proponente (VE-1.1).
\end{itemize}

% UC1 - Scrittura di testo nell'editor
\hypertarget{uc1}{}
\subsubsection{UC 1: Scrittura testo nell'editor}

% use case diagram
\begin{figure}[H]
    \centering
    \includegraphics[width=0.5\textwidth]{scrittura_nota.png}
    \caption{UC 1 – Scrittura di testo nell'editor}
\end{figure}

\begin{itemize}
    \item \textbf{Attore Principale}: Utente
    \item \textbf{Precondizioni}: L'editor è attivo ed è presente una nota aperta nell'editor.
    \item \textbf{Scenario Principale\textsubscript{G}}:
    \begin{enumerate}
        \item L'utente posiziona il cursore nell'area di testo.
        \item L'utente inserisce testo tramite tastiera.
        \item Il sistema aggiorna in tempo reale l'area di testo con il contenuto inserito.
        \item Il sistema aggiorna l'anteprima renderizzata, se la modalità di visualizzazione attiva lo prevede.
    \end{enumerate}
    \item \textbf{Postcondizioni}: Il testo inserito è visibile nell'editor e, ove applicabile, nell'anteprima renderizzata.
    \item \textbf{Trigger}: L'utente vuole inserire o modificare i contenuti testuali della nota.
\end{itemize}
